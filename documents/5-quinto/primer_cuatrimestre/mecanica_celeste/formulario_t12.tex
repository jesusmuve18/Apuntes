\begin{bx}{Tema 1}
\begin{minipage}[t]{0.485\linewidth}
\textbf{Fuerza y ecuación de movimiento}
\[F(x)=-\frac{GMm}{|x|^2}\frac{x}{|x|},\quad x\in\R^3\setminus\{0\}\]
\[\mu=GM,\qquad f(r)=-\frac{\mu}{r^2}\]
\[F(x)=f(|x|)\frac{x}{|x|}\ \ (\text{campo de fuerzas}),\qquad f:(0,+\infty)\to\R\]
\[\ddot x(t)=f\big(|x(t)|\big)\frac{x(t)}{|x(t)|}\]
\textbf{Energía}
\[E_c=\tfrac12 mv^2,\qquad E_{pg}=mgh\]
\[F=-\nabla V\ \text{ en }\ \Omega\subseteq\R^3\ \text{abierto}\]
\[E=\frac{|\dot x|^2}{2}+V(x)=\frac{|\dot x|^2}{2}-\Phi(|x|),\quad \Phi(r)=\int_1^r f(s)\,ds\]
\[E=\frac{|\dot x|^2}{2}-\frac{\mu}{|x|}\quad(\text{caso newtoniano})\]
\end{minipage}\hfill\vrule\hfill
\begin{minipage}[t]{0.485\linewidth}
\textbf{Momento angular y área}
\[c(t)=x(t)\wedge\dot x(t)=r^2\dot\theta\,(0,0,1)\]
\[x=r(\cos\theta,\sen\theta)\ \Rightarrow\ \dot\theta=x_1\dot x_2-\dot x_1x_2\]
\[\text{Área}(D)=\frac12\int_{t_0}^{t_1}r^2\dot\theta\,dt=\frac12|c|\,(t_1-t_0)\]
\vspace{1cm}
\begin{key}\centering\textbf{TEOREMA DE PROLONGACIÓN} $(c=0)$\\[2pt]
\raggedright
i) $\displaystyle\lim_{t\to\omega}\ r(t)^2+\dot r(t)^2=+\infty$\qquad
ii) $\exists\,t_n\to\omega:\ r(t_n)\to0$
\end{key}
\end{minipage}
\end{bx}
